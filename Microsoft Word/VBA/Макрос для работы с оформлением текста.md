# Макрос для работы с оформлением текста
Описание: мой шаблон листа с макросами в Normal.dot

```vba
Option Explicit

Sub Добавляет_символ_умножения() ' Добавляет символ умножения [Ctrl+Shift+*]
Selection.InsertSymbol CharacterNumber:=8729, Unicode:=True, Bias:=0
End Sub

Sub SpaceAfter_Up() ' Увеличить интервал после [Ctrl+Num8]
With Selection.ParagraphFormat
If .SpaceAfter < 1583 Then
.SpaceAfter = .SpaceAfter + 1
End If
End With
End Sub

Sub SpaceAfter_Down() ' Уменьшить интервал после [Ctrl+Num2]
With Selection.ParagraphFormat
If .SpaceAfter > 0 Then
.SpaceAfter = .SpaceAfter - 1
End If
End With
End Sub

Sub Макрос2() ' Сбросить все интервалы [Ctrl+Num5]
With Selection.ParagraphFormat
.LeftIndent = CentimetersToPoints(0)
.RightIndent = CentimetersToPoints(0)
.SpaceBefore = 0
.SpaceBeforeAuto = False
.SpaceAfter = 0
.SpaceAfterAuto = False
.LineUnitBefore = 0
.LineUnitAfter = 0
End With
End Sub

Sub UPSizeCase() ' Увеличить величину шрифта [Ctrl+Num7]
If Selection.Font.Size < 1638 Then
Selection.Font.Size = Selection.Font.Size + 0.5
End If
End Sub

Sub DownSizeCase() ' Уменьшить величину шрифта [Ctrl+Num1]
If Selection.Font.Size > 1 Then
Selection.Font.Size = Selection.Font.Size - 0.5
End If
End Sub

Sub Spacing_Up() 'Увеличить межзнаковый интервал [Ctrl+Num9]
With Selection.Font
If .Spacing < 1584 Then
.Spacing = .Spacing + 0.1
End If
End With
End Sub

Sub Spacing_Down() ' Уменьшить межзнаковый интервал [Ctrl+Num3]
With Selection.Font
If .Spacing < 1584 Then
.Spacing = .Spacing - 0.1
End If
End With
End Sub

Sub Line_spacing_exactly_Up() ' Увеличить междустрочный интервал (Точно) [Ctrl+Num6]
With Selection.ParagraphFormat
.LineSpacingRule = wdLineSpaceExactly
.LineSpacing = .LineSpacing + 0.1
End With
End Sub

Sub Line_spacing_exactly_Down() ' Уменьшить междустрочный интервал (Точно) [Ctrl+Num4]
With Selection.ParagraphFormat
.LineSpacingRule = wdLineSpaceExactly
.LineSpacing = .LineSpacing - 0.1
End With
End Sub

Sub sub_doc_statistics() 'Отобразить статистику документа
Dim doc As Document
Dim statsInfo As String

Set doc = ActiveDocument

statsInfo = "Создан: " & vbTab & vbTab & vbTab & vbTab & doc.BuiltInDocumentProperties("Creation Date") & vbCrLf
On Error Resume Next
statsInfo = statsInfo & "Сохранен: " & vbTab & vbTab & vbTab & doc.BuiltInDocumentProperties("Last Save Time") & vbCrLf
On Error GoTo 0
statsInfo = statsInfo & vbCrLf
statsInfo = statsInfo & "Кол-во страниц: " & vbTab & vbTab & vbTab & doc.ComputeStatistics(wdStatisticPages) & vbCrLf & vbCrLf
statsInfo = statsInfo & "Кол-во слов: " & vbTab & vbTab & vbTab & doc.ComputeStatistics(wdStatisticWords) & vbCrLf
statsInfo = statsInfo & "Кол-во символов с пробелами: " & vbTab & doc.ComputeStatistics(wdStatisticCharactersWithSpaces) & vbCrLf
statsInfo = statsInfo & "Кол-во символов без пробелов: " & vbTab & doc.ComputeStatistics(wdStatisticCharacters) & vbCrLf
statsInfo = statsInfo & "Кол-во абзацев: " & vbTab & vbTab & vbTab & doc.ComputeStatistics(wdStatisticParagraphs) & vbCrLf
statsInfo = statsInfo & "Кол-во строк: " & vbTab & vbTab & vbTab & doc.ComputeStatistics(wdStatisticLines)
MsgBox statsInfo, vbInformation, "Статистика документа"
End Sub
```
